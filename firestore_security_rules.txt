rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ========================
    // HELPERS
    // ========================

    function signedIn() {
      return request.auth != null;
    }

    function userDoc() {
      return get(
        /databases/$(database)/documents/users/$(request.auth.uid)
      );
    }

    function userExists() {
      return exists(
        /databases/$(database)/documents/users/$(request.auth.uid)
      );
    }

    function myRole() {
      // Safely get role, default to empty string if document or field missing
      return userDoc().data.role;
    }

    function isAdmin() {
      return signedIn() &&
        userExists() &&
        myRole() in ['admin', 'superAdmin', 'super_admin', 'superadmin'];
    }

    function isSuperAdmin() {
      return signedIn() &&
        userExists() &&
        myRole() in ['superAdmin', 'super_admin', 'superadmin'];
    }

    function isStudent() {
      // Anonymous auth users are students if they have no user doc
      // OR if their user doc has role == 'student'
      return signedIn() && (
        !userExists() ||
        myRole() == 'student'
      );
    }

    function noticeOwner(noticeId) {
      // Check if the authenticated user is the author of the notice
      // The field in Firestore is 'authorId' (matching the NoticeModel)
      return get(/databases/$(database)/documents/notices/$(noticeId)).data.authorId == request.auth.uid;
    }

    // ========================
    // USERS
    // ========================

    match /users/{userId} {

      allow read: if signedIn();

      allow create: if signedIn()
        && (
          request.auth.uid == userId ||
          isSuperAdmin()
        );

      allow update: if signedIn()
        && (
          request.auth.uid == userId ||
          isAdmin()
        );

      allow delete: if isSuperAdmin();
    }

    // ========================
    // NOTICES
    // ========================

    match /notices/{noticeId} {

      // Anyone can read approved notices.
      // isAdmin() handles safety for user doc existence.
      allow read: if resource.data.status == 'approved'
        || isAdmin()
        || isStudent();

      // Only admins create notices
      allow create: if isAdmin();

      // Super admin can update ANY notice (approve, unapprove, reject, edit)
      // A regular admin can update/approve notices they created (their notices)
      // Notice owner (admin who created it) can also update
      allow update: if signedIn()
        && (
          isSuperAdmin() ||
          isAdmin() ||
          (resource.data.authorId == request.auth.uid)
        );

      // Super admin, admins, or the notice owner can delete notices
      allow delete: if signedIn()
        && (
          isSuperAdmin() ||
          isAdmin() ||
          (resource.data.authorId == request.auth.uid)
        );
    }

    // ========================
    // LIKES (Student UID-based for Anonymous Auth)
    // ========================

    match /likes/{likeId} {

      // Only authenticated users can read likes
      allow read: if signedIn();

      // Create a like - studentUid must match the authenticated user's UID
      // Anonymous users have auth.uid set by Firebase, so this works.
      allow create: if signedIn()
        && request.resource.data.noticeId is string
        && request.resource.data.studentUid == request.auth.uid
        && request.resource.data.createdAt is timestamp;

      // Delete a like - must own the like (studentUid matches)
      allow delete: if signedIn()
        && resource.data.studentUid == request.auth.uid;

      // No updates allowed (toggle = create + delete)
      allow update: if false;
    }

    // ========================
    // FEEDBACK
    // ========================

    match /feedback/{feedbackId} {

      allow create: if signedIn()
        && request.resource.data.userId == request.auth.uid;

      allow read: if signedIn()
        && (
          resource.data.userId == request.auth.uid ||
          isAdmin()
        );

      allow update: if isAdmin();

      allow delete: if signedIn()
        && (
          resource.data.userId == request.auth.uid ||
          isSuperAdmin()
        );
    }

    // ========================
    // FAQS
    // ========================

    match /faqs/{faqId} {

      allow read: if true;

      allow create, update, delete: if isAdmin();
    }

    // ========================
    // NOTIFICATIONS
    // ========================

    match /notifications/{notificationId} {

      allow read: if signedIn();

      // Super admin can create notifications too (for notification system actions)
      allow create: if isSuperAdmin() || isAdmin();

      allow update, delete: if isSuperAdmin();
    }
  }
}
